library(shiny)

ui <- fluidPage(
  titlePanel("My App"),                 # Title at top
  sidebarLayout(
    sidebarPanel(
      sliderInput("num", "Choose number:", 
                  min = 1, max = 100, value = 50)
    ),
    mainPanel(
      plotOutput("plot")                # Where plot appears
    )
  )
)

server <- function(input, output, session) {

  output$plot <- renderPlot({
    hist(rnorm(input$num))              # Uses input$num reactively
  })

}

shinyApp(ui, server)                    # Runs the app
